    Console

    zenml.console

    ZenML console implementation.

    Attributes

    console = Console(theme=zenml_custom_theme, markup=True) module-attribute

    error_console = Console(stderr=True, theme=zenml_custom_theme) module-attribute

    zenml_custom_theme = Theme(zenml_style_defaults) module-attribute

    zenml_style_defaults = {'info': Style(color='white', dim=True), 'warning': Style(color='yellow'), 'danger': Style(color='red', bold=True), 'title': Style(color='cyan', bold=True, underline=True), 'error': Style(color='red'), 'success': Style(color='green'), 'repr.str': Style(color='white', dim=False), 'repr.uuid': Style(color='magenta', dim=False)} module-attribute
